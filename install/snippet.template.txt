# Bluebox Kubernetes OTel collector: install (one command per line).
# Works unchanged in bash, zsh, fish, PowerShell and cmd: no variables, pipes,
# line continuations or shell quoting beyond plain double quotes.
# Placeholders the wizard fills: <REF> (commit of a values-vX.Y.Z tag),
# <DISTRO> (aks|eks|gke|local), <CLUSTER_NAME>, <ENDPOINT>, <PSA_LEVEL>
# (restricted, or privileged only when pod logs are on), <AGENT_EXTRA_FILES>
# and <CLUSTER_EXTRA_FILES> (" -f <BASE>/overlays/autopilot/agent.yaml",
# " -f <BASE>/pod-logs.agent.yaml", " -f <BASE>/overlays/ai/agent.yaml", ...),
# <AGENT_EXTRA_SETS> (" --set \"global.bluebox.podLogs.namespaces={shop,payments}\"").
# <BASE> = https://raw.githubusercontent.com/pawelsiwek/bluebox-k8s-collector/<REF>/values

helm repo add open-telemetry https://open-telemetry.github.io/opentelemetry-helm-charts --force-update
kubectl create namespace bluebox-otelcol
kubectl label namespace bluebox-otelcol pod-security.kubernetes.io/enforce=<PSA_LEVEL> pod-security.kubernetes.io/warn=<PSA_LEVEL> pod-security.kubernetes.io/audit=<PSA_LEVEL> --overwrite

# Token: run the next command and paste the Secret block below it. Replace
# <TOKEN> with the token that the wizard's "Show token" reveals, then press
# Ctrl-D (Ctrl-Z then Enter on Windows).
# The token never appears on a command line or in shell history.
kubectl apply -n bluebox-otelcol -f -
apiVersion: v1
kind: Secret
metadata:
  name: bluebox-otelcol-token
type: Opaque
stringData:
  token: <TOKEN>

helm upgrade --install bluebox-otelcol-agent open-telemetry/opentelemetry-collector --version 0.177.0 --namespace bluebox-otelcol --wait --atomic --timeout 10m -f <BASE>/common.yaml -f <BASE>/agent.yaml -f <BASE>/overlays/<DISTRO>/agent.yaml<AGENT_EXTRA_FILES> --set "global.bluebox.clusterName=<CLUSTER_NAME>" --set "global.bluebox.endpoint=<ENDPOINT>"<AGENT_EXTRA_SETS>
helm upgrade --install bluebox-otelcol-cluster open-telemetry/opentelemetry-collector --version 0.177.0 --namespace bluebox-otelcol --wait --atomic --timeout 10m -f <BASE>/common.yaml -f <BASE>/cluster.yaml -f <BASE>/overlays/<DISTRO>/cluster.yaml<CLUSTER_EXTRA_FILES> --set "global.bluebox.clusterName=<CLUSTER_NAME>" --set "global.bluebox.endpoint=<ENDPOINT>"

# Verify: prints otlp_status, cause, ready and counts; paste the output into the wizard.
helm test bluebox-otelcol-agent --namespace bluebox-otelcol --logs
